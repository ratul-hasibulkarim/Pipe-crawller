# Native unit tests without PlatformIO (pio test -e native runs the same files).
#   cmake -S firmware/test -B firmware/build/test && cmake --build firmware/build/test && ctest --test-dir firmware/build/test
cmake_minimum_required(VERSION 3.16)
project(pipebot_fw_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
set(FW ${CMAKE_CURRENT_SOURCE_DIR}/..)
file(GLOB CORE_SRC ${FW}/src/control/*.cpp ${FW}/src/comm/*.cpp ${FW}/src/core/*.cpp ${FW}/src/sim/*.cpp)
add_library(unity STATIC ${FW}/third_party/unity/unity.c)
target_include_directories(unity PUBLIC ${FW}/third_party/unity)
target_compile_definitions(unity PUBLIC UNITY_INCLUDE_DOUBLE)
file(GLOB TEST_DIRS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/test_*)
foreach(SIZE S M L)
  add_library(core_${SIZE} STATIC ${CORE_SRC})
  target_include_directories(core_${SIZE} PUBLIC ${FW}/include ${FW}/src)
  target_compile_definitions(core_${SIZE} PUBLIC PIPEBOT_SIZE_${SIZE})
  target_compile_options(core_${SIZE} PRIVATE -Wall -Wextra -Werror)
  foreach(T ${TEST_DIRS})
    add_executable(${T}_${SIZE} ${CMAKE_CURRENT_SOURCE_DIR}/${T}/test_main.cpp)
    target_link_libraries(${T}_${SIZE} core_${SIZE} unity)
    add_test(NAME ${T}_${SIZE} COMMAND ${T}_${SIZE})
  endforeach()
endforeach()
